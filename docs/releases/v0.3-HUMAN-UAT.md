**U8 — Activar webhook y preparar la entrega firmada**
En la ficha del servicio: **Enable webhook**; copiar payload URL y secreto (se muestran una sola vez).
```sh
export GH_REPO=<gh-user>/noodara-uat-node-api HOOK_URL='<payload-url>'; read -rs HOOK_SECRET
deliver() { sha=$(git ls-remote https://github.com/$GH_REPO refs/heads/main | cut -f1)
  body=$(printf '{"ref":"refs/heads/main","after":"%s","deleted":false,"repository":{"clone_url":"https://github.com/%s.git","ssh_url":"git@github.com:%s.git","html_url":"https://github.com/%s"},"head_commit":{"id":"%s"}}' "$sha" "$GH_REPO" "$GH_REPO" "$GH_REPO" "$sha")
  sig=$(printf '%s' "$body" | openssl dgst -sha256 -hmac "$HOOK_SECRET" | sed 's/^.* //')
  curl -sS -w '\nHTTP %{http_code}\n' -X POST "$HOOK_URL" -H 'content-type: application/json' -H 'X-GitHub-Event: push' \
    -H "X-GitHub-Delivery: $(cat /proc/sys/kernel/random/uuid)" -H "X-Hub-Signature-256: sha256=$sig" --data-binary "$body"; }
q "select name, webhook_hook_id is not null hook, webhook_secret_credential_id is not null secret from services where name='node-api-v03'"
```
