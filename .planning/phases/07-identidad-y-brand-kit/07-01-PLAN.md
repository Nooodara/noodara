Output: `packages/ui/src/brand/geometry.ts`, `glyphs.ts`, `concepts/a.ts`, `concepts/b.ts`, `concepts/c.ts`, and `geometry.test.ts` (green).

```typescript
export const GRID = 24;                 // design grid, D-15 (never change)
export const MARGIN = 1;                // grid units of breathing room inside the 24 box
export const STROKE = 4;                // stem/diagonal thickness in grid units (starting value; adjustable in D-16 rounds)
export const APERTURE_RADIUS = 5;       // outer radius of the aperture ring AND of the wordmark "o" (D-08)
export const TERMINAL_RADIUS = STROKE / 2;
export const X_HEIGHT = 12;             // wordmark x-height in grid units
export const ASCENDER = 18;             // wordmark ascender height (d) in grid units
export const LETTER_GAP = 2;            // side bearing between glyphs (D-07 tight tracking)
export const LOCKUP_GAP = 6;            // gap between monogram and wordmark in the lockup

export const CONCEPT_IDS = ['a', 'b', 'c'] as const;
export type ConceptId = (typeof CONCEPT_IDS)[number];
export const DEFAULT_CONCEPT: ConceptId; // 'a' until 07-05 records the approved concept

export type MonogramPartName = 'stem-left' | 'diagonal' | 'stem-right' | 'aperture' | 'frame';
export interface PathPart { readonly part: MonogramPartName | `glyph-${string}`; readonly d: string; }
export interface ConceptMeta { readonly id: ConceptId; readonly name: string; readonly meaning: string; }

export const CONCEPT_META: Readonly<Record<ConceptId, ConceptMeta>>;

export function monogramParts(concept: ConceptId): readonly PathPart[];
export function monogramPath(concept: ConceptId): string;          // parts joined by a single space
export function wordmarkParts(concept: ConceptId): readonly PathPart[];
export function wordmarkPath(concept: ConceptId): string;
export function wordmarkWidth(concept: ConceptId): number;          // in grid units
export interface LockupLayout { readonly width: number; readonly height: number; readonly monogramX: number; readonly wordmarkX: number; readonly wordmarkY: number; }
export function lockupLayout(concept: ConceptId): LockupLayout;
export function lockupParts(concept: ConceptId): { readonly monogram: readonly PathPart[]; readonly wordmark: readonly PathPart[] };

// primitives (exported so tests and the construction sheet can reuse them)
export function bar(x: number, y: number, w: number, h: number): string;
export function ring(cx: number, cy: number, rOuter: number, rInner: number): string;   // evenodd ring, arcs only
export function diagonalBar(x1: number, y1: number, x2: number, y2: number, width: number): string;
export function fmt(n: number): string;   // fixed 3-decimal max, trailing zeros stripped, "-0" -> "0"
```
